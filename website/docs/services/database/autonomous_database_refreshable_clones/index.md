```sql
SELECT
id,
region
FROM oci.database.autonomous_database_refreshable_clones
WHERE autonomousDatabaseId = '{{ autonomousDatabaseId }}' -- required
AND region = '{{ region }}' -- required
AND opc-request-id = '{{ opc-request-id }}'
AND limit = '{{ limit }}'
AND page = '{{ page }}'
;
```
