```sql
SELECT
id,
associatedDatabases,
associatedLongTermBackupCount,
associatedLongTermBackups,
compartmentId,
definedTags,
displayName,
freeformTags,
lifecycleDetails,
lifecycleState,
systemTags,
timeCreated,
typeDetails
FROM oci.database.key_stores
WHERE keyStoreId = '{{ keyStoreId }}' -- required
AND region = '{{ region }}' -- required
AND opc-request-id = '{{ opc-request-id }}'
;
```
